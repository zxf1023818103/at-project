cmake_minimum_required(VERSION 3.18)
find_package(bouffalo_sdk REQUIRED HINTS $ENV{BL_SDK_BASE})
sdk_add_include_directories(.)
sdk_set_main_file(main.c)
include(nan_memory.cmake)
project(nan_host)
set(AT_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../..)
set_target_properties(${proj_name}.elf PROPERTIES LINK_FLAGS "-T${CMAKE_CURRENT_BINARY_DIR}/generated/linker.ld -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/build_out/${proj_name}.map")
add_compile_definitions(AT_GREEN2_NAN CONFIG_WIFI_NAN_HOST_APP)
add_library(FreeRTOS ALIAS libfreertos)
add_library(freertos_kernel ALIAS libfreertos)
add_library(freertos_config INTERFACE)
target_link_libraries(freertos_config INTERFACE sdk_intf_lib)
target_include_directories(freertos_config INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}
    ${AT_ROOT}/component/freertos-plus-tcp_config)
add_subdirectory(${AT_ROOT}/component/bluedroid_tcp_config ${CMAKE_CURRENT_BINARY_DIR}/freertos_tcp)
add_library(monitor STATIC ${AT_ROOT}/component/freertos-cli_config/freertos-cli/FreeRTOS_CLI.c)
target_include_directories(monitor PUBLIC ${AT_ROOT}/component/freertos-cli_config/freertos-cli)
target_link_libraries(monitor PUBLIC FreeRTOS freertos_config)
include(mbedtls_sdk.cmake)
add_subdirectory(${AT_ROOT}/component/mbedtls_config ${CMAKE_CURRENT_BINARY_DIR}/mbedtls_config)
foreach(t mbedcrypto mbedx509 mbedtls p256m everest)
    if(TARGET ${t})
        target_include_directories(${t} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${AT_ROOT}/component/mbedtls_config)
        target_compile_definitions(${t} PUBLIC MBEDTLS_USER_CONFIG_FILE="nan_mbedtls_config.h"
            MBEDTLS_THREADING_ALT MBEDTLS_ENTROPY_HARDWARE_ALT)
        target_link_libraries(${t} PUBLIC freertos_config)
    endif()
endforeach()
include(${AT_ROOT}/component/amebagreen2/nan/host.cmake)
if(CONFIG_NAN_USB_DIAGNOSTICS)
    set(AT_NAN_NET_DIAG ON)
endif()
nan_add_freertos_host(bl618dg_nan)
target_compile_definitions(bl618dg_nan PRIVATE NAN_DEVICE_NAME="BL618DG FreeRTOS")
target_include_directories(bl618dg_nan PRIVATE ${AT_ROOT}/component/freertos-plus-tcp_config)
if(CONFIG_NAN_NATIVE_WHC_USB)
    add_subdirectory(${AT_ROOT}/component/cherryusb_config/whc_usb ${CMAKE_CURRENT_BINARY_DIR}/whc_usb)
    set(NAN_TRANSPORT whc_usb)
    target_sources(${proj_name}.elf PRIVATE nan_native_port.c nan_store.c)
else()
    add_subdirectory(${AT_ROOT}/component/cherryusb_config/nan_usb ${CMAKE_CURRENT_BINARY_DIR}/nan_usb)
    target_compile_definitions(nan_usb PUBLIC NAN_USB_HOST=1)
    target_link_libraries(nan_usb PRIVATE sdk_intf_lib)
    set(NAN_TRANSPORT nan_usb)
endif()
include(${AT_ROOT}/project/bl618dg_usb_bench/app/ehci_completion.cmake)
if(CONFIG_NAN_USB_DIAGNOSTICS)
    include(native_ehci_diag.cmake)
    target_sources(${proj_name}.elf PRIVATE nan_rx_perf.c)
    foreach(symbol green2_nan_rx at_netif_rx_alloc FreeRTOS_MatchingEndpoint
                   xSendEventStructToIPTask green2_nan_tcp_command prvSendData)
        target_link_options(${proj_name}.elf PRIVATE -Wl,--wrap=${symbol})
    endforeach()
    # Rename memcpy only in this shared source's diagnostic compilation.
    # Flash HAL/IRAM memcpy calls remain untouched, including during XIP-off.
    set_property(SOURCE ${AT_ROOT}/component/amebagreen2/nan/nan_network.c
        APPEND PROPERTY COMPILE_DEFINITIONS memcpy=bl618dg_nan_profile_memcpy)
endif()
target_sources(${proj_name}.elf PRIVATE nan_port.c nan_protocol_selftest.c nan_heap.c)
target_include_directories(${proj_name}.elf PRIVATE ${AT_ROOT}/component/amebagreen2/nan/vendor)
target_link_libraries(${proj_name}.elf bl618dg_nan ${NAN_TRANSPORT} mbedtls_config freertos_plus_tcp monitor)
target_compile_options(${proj_name}.elf PRIVATE -O2)
# SDK heap_3 pins pvPortMalloc to OCRAM. Route CPU-only protocol and packet
# allocations to the selected data heap; kernel stacks and CherryUSB DMA keep
# their original allocators. vPortFree uses kfree and accepts either region.
foreach(t ${proj_name}.elf bl618dg_nan ${NAN_TRANSPORT} freertos_plus_tcp freertos_plus_tcp_port)
    target_compile_definitions(${t} PRIVATE pvPortMalloc=bl618dg_nan_malloc)
endforeach()
