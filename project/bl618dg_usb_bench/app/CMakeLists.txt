cmake_minimum_required(VERSION 3.18)
find_package(bouffalo_sdk REQUIRED HINTS $ENV{BL_SDK_BASE})
sdk_add_include_directories(.)
sdk_set_main_file(main.c)
project(usb_bench)
set_target_properties(${proj_name}.elf PROPERTIES LINK_FLAGS "-T${CMAKE_CURRENT_BINARY_DIR}/generated/linker.ld -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/build_out/${proj_name}.map")
set(bench ${CMAKE_CURRENT_LIST_DIR}/../../../component/cherryusb_config/benchmark)
target_sources(${proj_name}.elf PRIVATE ${bench}/usb_bench.c)
target_include_directories(${proj_name}.elf PRIVATE ${bench})
target_compile_definitions(${proj_name}.elf PRIVATE USB_BENCH_BOARD="bl618dg")
if(CONFIG_USB_BENCH_HOST)
    target_compile_definitions(${proj_name}.elf PRIVATE USB_BENCH_HOST=1)
else()
    target_compile_definitions(${proj_name}.elf PRIVATE USB_BENCH_HOST=0)
endif()
target_compile_options(${proj_name}.elf PRIVATE -O2)

if(CONFIG_USB_BENCH_HOST)
    include(${CMAKE_CURRENT_LIST_DIR}/ehci_completion.cmake)
endif()
