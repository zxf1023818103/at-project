cmake_minimum_required(VERSION 3.15)
set(WIFI6_NET_AL_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../component/bouffalo_wifi_tcp_config")
find_package(bouffalo_sdk REQUIRED HINTS $ENV{BL_SDK_BASE})
sdk_add_include_directories(.)
# Preserve the KV namespace used by the existing at-project SDK fork.
sdk_add_compile_definitions(-DLFS_KV_NAMESPACE=\"/KV\")
if(CONFIG_PAN_RF_DIAG)
    sdk_set_main_file(main_rf_diag.c)
elseif(CONFIG_PAN_WIFI_ONLY)
    sdk_set_main_file(main_wifi.c)
elseif(CONFIG_PAN_HOST)
    sdk_set_main_file(main_bluedroid.c)
else()
    sdk_set_main_file(main.c)
endif()
sdk_add_compile_definitions(-DCONFIG_BOOT_NP_ASYNC)
set(BT_PORT ${BL_SDK_BASE}/components/wireless/bluetooth/btblecontroller/btblecontroller_port)
sdk_add_include_directories(${BT_PORT})
project(pan)
# SDK assumes BUILD_DIR is relative when setting -T; use the actual binary dir.
set_target_properties(${proj_name}.elf PROPERTIES LINK_FLAGS "-T${CMAKE_CURRENT_BINARY_DIR}/generated/linker.ld -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/build_out/${proj_name}.map")
if(NOT CONFIG_PAN_WIFI_ONLY OR CONFIG_PAN_RF_DIAG)
    target_sources(${proj_name}.elf PRIVATE ${BT_PORT}/btblecontroller_port.c ${BT_PORT}/btblecontroller_port_freertos_os.c)
endif()
set_source_files_properties(${BT_PORT}/btblecontroller_port.c PROPERTIES COMPILE_OPTIONS "-include;bl618dg_mfg_media.h")
if(CONFIG_PAN_HOST OR CONFIG_PAN_WIFI_ONLY)
    add_library(freertos_config INTERFACE)
    target_link_libraries(freertos_config INTERFACE sdk_intf_lib)
    target_include_directories(freertos_config INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
    add_library(freertos_kernel ALIAS libfreertos)
    add_subdirectory(../../../component/bluedroid_tcp_config ${CMAKE_CURRENT_BINARY_DIR}/freertos_tcp)
endif()
if(NOT CONFIG_PAN_WIFI_ONLY)
    add_subdirectory(../../../component/bluedroid_config ${CMAKE_CURRENT_BINARY_DIR}/bluedroid_config)
endif()
if(CONFIG_PAN_SPP)
    target_sources(${proj_name}.elf PRIVATE spp_load.c)
endif()
if(CONFIG_PAN_HOST OR CONFIG_PAN_WIFI_ONLY)
    target_sources(${proj_name}.elf PRIVATE pan_network.c)
    target_sources(${proj_name}.elf PRIVATE pan_load.c)
    if(CONFIG_PAN_WIFI_ONLY)
        target_link_libraries(${proj_name}.elf freertos_plus_tcp)
    else()
        target_link_libraries(${proj_name}.elf bluedroid_pan_tcp)
    endif()
    if(CONFIG_PAN_COEX OR CONFIG_PAN_WIFI_ONLY)
        target_sources(${proj_name}.elf PRIVATE pan_coex.c)
    endif()
endif()
