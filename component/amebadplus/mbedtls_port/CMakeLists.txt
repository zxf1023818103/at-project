add_library(mbedtls_cfg INTERFACE)
target_compile_definitions(mbedtls_cfg INTERFACE
    MBEDTLS_USER_CONFIG_FILE="mbedtls_user_config.h"
    MBEDTLS_THREADING_ALT MBEDTLS_ENTROPY_HARDWARE_ALT)
target_link_libraries(mbedtls_cfg INTERFACE mbedtls_config_headers)
foreach(t mbedcrypto mbedx509 mbedtls p256m everest)
    if(TARGET ${t})
        target_link_libraries(${t} PUBLIC mbedtls_cfg)
        target_include_directories(${t} BEFORE PRIVATE ${AT_ROOT}/component/syscall_config)
        target_compile_options(${t} PRIVATE -w)
    endif()
endforeach()
add_library(mbedtls_port STATIC entropy.c)
target_link_libraries(mbedtls_port PRIVATE amebadplus_sdk)
target_link_libraries(mbedtls_config mbedtls_port)
if(AT_ENABLE_HWGCM)
    # Retain the complete upstream streaming implementation as a private fallback.
    # Only build-local copies are renamed; the shared mbedTLS checkout is untouched.
    set(_gcm_gen ${CMAKE_CURRENT_BINARY_DIR}/gcm)
    file(MAKE_DIRECTORY ${_gcm_gen})
    foreach(part include/mbedtls/gcm.h library/gcm.c)
        file(READ ${AT_ROOT}/component/mbedtls_config/mbedtls/${part} data)
        string(REPLACE "mbedtls_gcm_" "dplus_sw_gcm_" data "${data}")
        string(REPLACE "dplus_sw_gcm_context" "mbedtls_gcm_context" data "${data}")
        if(part MATCHES "[.]h$")
            string(REPLACE "MBEDTLS_GCM_H" "DPLUS_SW_GCM_H" data "${data}")
            file(WRITE ${_gcm_gen}/dplus_gcm_sw.h "${data}")
        else()
            string(REPLACE "\"mbedtls/gcm.h\"" "\"dplus_gcm_sw.h\"" data "${data}")
            string(REPLACE "#if !defined(MBEDTLS_GCM_ALT)" "#if 1" data "${data}")
            file(WRITE ${_gcm_gen}/dplus_gcm_sw.c "${data}")
        endif()
    endforeach()
    target_compile_definitions(mbedtls_cfg INTERFACE MBEDTLS_GCM_ALT)
    target_include_directories(mbedtls_cfg INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
    target_sources(mbedcrypto PRIVATE gcm_alt.c ${_gcm_gen}/dplus_gcm_sw.c)
    target_include_directories(mbedcrypto PRIVATE ${_gcm_gen}
        ${AT_ROOT}/component/mbedtls_config/mbedtls/library)
    target_link_libraries(mbedcrypto PRIVATE amebadplus_sdk FreeRTOS)
endif()
