# component/iperf3_config — upstream esnet/iperf (iperf3) wrapped as a component, mirroring the
# littlefs_config / coresntp_config recipe (submodule + explicit source list + INTERFACE umbrella).
# Submodule: component/iperf3_config/iperf3 (pinned to tag 3.21).
#
# iperf3 is a POSIX/BSD-sockets program; this target only has FreeRTOS-Plus-TCP's FreeRTOS_* sockets
# (no lwIP, no BSD socket()), so building the `iperf3` lib here SURFACES the porting gaps (missing
# <sys/socket.h>/<netinet/*>/<netdb.h>, getaddrinfo, select/poll, pthread, …) rather than producing a
# working lib yet. The shims are a follow-on once the gap list is in.
#
# Targets:
#   iperf3         STATIC EXCLUDE_FROM_ALL — the libiperf core (no SCTP, no CLI main, no unit tests).
#   iperf3_config  INTERFACE umbrella consumers link.
#
# cJSON: iperf bundles its own src/cjson.c (a cJSON patched to int64_t valueint). We do NOT build it —
# we reuse component/cjson_config (DaveGamble cJSON). iperf's sources #include "cjson.h" (its bundled
# header, still int64_t valueint); reconciling that struct ABI with cjson_config's standard `int
# valueint` (a 1-line iperf3/src/cjson.h -> #include "cJSON.h" redirect) is the runtime fix applied
# when the lib is actually wired up — it is not needed for this compile pass.

set(_src "${CMAKE_CURRENT_SOURCE_DIR}/iperf3/src")

# iperf_config.h — normally produced by autotools ./configure. Generate it with configure_file() from
# the upstream autoheader template: that template is `#undef SYMBOL` lines (which configure_file would
# leave verbatim -> PACKAGE_VERSION/etc. undefined -> compile errors), so transform `#undef X` ->
# `#cmakedefine X @X@` first, then define only what a bare-metal threaded build needs and leave the
# rest undefined (the autoconf "feature absent" default).
file(READ  "${_src}/iperf_config.h.in" _tpl)
string(REGEX REPLACE "#undef[ \t]+([A-Za-z0-9_]+)" "#cmakedefine \\1 @\\1@" _tpl "${_tpl}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/iperf_config.h.in.cmake" "${_tpl}")
set(PACKAGE        "\"iperf\"")
set(PACKAGE_NAME   "\"iperf\"")
set(PACKAGE_TARNAME "\"iperf\"")
set(PACKAGE_VERSION "\"3.21\"")
set(VERSION        "\"3.21\"")
set(PACKAGE_STRING "\"iperf 3.21\"")
set(HAVE_PTHREAD 1)                      # iperf 3.16+ is mandatorily threaded
set(HAVE_POLL_H 1)                       # we ship a <poll.h> shim, so net.c may include it
configure_file("${CMAKE_CURRENT_BINARY_DIR}/iperf_config.h.in.cmake"
               "${CMAKE_CURRENT_BINARY_DIR}/iperf_config.h" @ONLY)

# version.h is normally git-generated by iperf's build; the only symbol used is IPERF_VERSION.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h" "#define IPERF_VERSION \"3.21\"\n")

add_library(iperf3 STATIC EXCLUDE_FROM_ALL
    "${_src}/iperf_api.c"        "${_src}/iperf_error.c"      "${_src}/iperf_auth.c"
    "${_src}/iperf_client_api.c" "${_src}/iperf_server_api.c" "${_src}/iperf_tcp.c"
    "${_src}/iperf_udp.c"        "${_src}/iperf_util.c"       "${_src}/iperf_time.c"
    "${_src}/iperf_pthread.c"    "${_src}/dscp.c"             "${_src}/net.c"
    "${_src}/iperf_locale.c"     "${_src}/tcp_info.c"         "${_src}/timer.c"
    "${_src}/units.c")
    # excluded: cjson.c (-> component/cjson_config), iperf_sctp.c (no <netinet/sctp.h>),
    #           main.c + t_*.c (the CLI program shell + upstream unit tests)
target_include_directories(iperf3 PUBLIC "${_src}" "${CMAKE_CURRENT_BINARY_DIR}")  # +generated iperf_config.h
# Compile-time include dirs ONLY (via genex), so iperf3 has NO target link deps — that lets the
# consumer drop libiperf3.a into its cross_refs --start-group as a raw archive without a CMake
# dependency cycle (its PUBLIC _config-umbrella deps would otherwise depend back on that group).
# Borrowed: the BSD-sockets/POSIX shim headers (component/syscall_config; impls force-linked from
# there) + FreeRTOS-Plus-TCP / FreeRTOS / cJSON headers. NOT linked here.
target_include_directories(iperf3 PRIVATE
    "${CMAKE_SOURCE_DIR}/component/syscall_config"
    "${CMAKE_SOURCE_DIR}/component/cjson_config/cjson"   # cJSON.h (iperf's cjson.h redirects here)
    $<TARGET_PROPERTY:freertos-plus-tcp_config,INTERFACE_INCLUDE_DIRECTORIES>
    $<TARGET_PROPERTY:freertos_config,INTERFACE_INCLUDE_DIRECTORIES>
    $<TARGET_PROPERTY:FreeRTOS,INTERFACE_INCLUDE_DIRECTORIES>)
target_compile_options(iperf3 PRIVATE -w)            # vendor warnings
# newlib's <pthread.h> declares the funcs but not the cancel-state constants iperf3 passes to
# pthread_setcancelstate/setcanceltype; define them here (the pthread shim ignores cancel state).
target_compile_definitions(iperf3 PRIVATE
    PTHREAD_CANCEL_ENABLE=0 PTHREAD_CANCEL_DEFERRED=0 PTHREAD_MUTEX_ERRORCHECK=1)

add_library(iperf3_config INTERFACE)
target_link_libraries(iperf3_config INTERFACE iperf3)
