cmake_minimum_required(VERSION 3.20)
project(green2_nan_tests C)
set(CMAKE_C_STANDARD 11)
add_compile_options(-ffunction-sections -fdata-sections -Wall -Wextra)
add_link_options(-Wl,--gc-sections)
option(NAN_TEST_SANITIZERS "Run native protocol tests with address/undefined sanitizers" OFF)
if(NAN_TEST_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../mbedtls_config/mbedtls mbedtls EXCLUDE_FROM_ALL)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(FreeRTOS INTERFACE)
target_include_directories(FreeRTOS INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/stubs)
set(GREEN2_NAN_NATIVE_TESTS ON)
add_subdirectory(../hostap hostap)
add_executable(crypto_test crypto_test.c test_port.c)
target_include_directories(crypto_test PRIVATE ../hostap ../hostap/utils ..)
target_compile_options(crypto_test PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/../pasn_endian.h)
target_compile_definitions(crypto_test PRIVATE CONFIG_SAE CONFIG_PASN CONFIG_NO_RANDOM_POOL CONFIG_NO_STDOUT_DEBUG)
target_link_libraries(crypto_test PRIVATE green2_pasn_raw mbedcrypto)
enable_testing()
add_test(NAME crypto_and_sae COMMAND crypto_test)
add_executable(pasn_test pasn_test.c test_port.c)
target_include_directories(pasn_test PRIVATE ../hostap ../hostap/utils ..)
target_compile_options(pasn_test PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/../pasn_endian.h)
target_compile_definitions(pasn_test PRIVATE CONFIG_SAE CONFIG_PASN CONFIG_IEEE80211W CONFIG_ECC CONFIG_NO_RANDOM_POOL CONFIG_NO_RADIUS CONFIG_NO_VLAN)
target_link_libraries(pasn_test PRIVATE green2_pasn_raw mbedcrypto)
add_test(NAME pasn_handshake COMMAND pasn_test)
add_executable(wire_test wire_test.c ../nan_wire.c)
target_include_directories(wire_test PRIVATE ..)
add_test(NAME secure_ndp_parser COMMAND wire_test)
add_executable(tcp_protocol_test tcp_protocol_test.c ../nan_tcp_protocol.c)
target_include_directories(tcp_protocol_test PRIVATE ..)
add_test(NAME tcp_throughput_protocol COMMAND tcp_protocol_test)
add_executable(auth_peer_test auth_peer_test.c ../fixes/whc_dev_nan_auth_peer_fix.c)
add_test(NAME pasn_early_peer_context COMMAND auth_peer_test)
add_executable(oob_test oob_test.c ../nan_oob.c ../nan_oob_protocol.c)
target_include_directories(oob_test PRIVATE oob_stubs .. ../vendor)
target_link_libraries(oob_test PRIVATE mbedcrypto)
add_test(NAME ble_oob_pin_lifecycle COMMAND oob_test)
add_executable(pasn_oob_test pasn_test.c ../nan_pasn.c test_port.c)
target_include_directories(pasn_oob_test PRIVATE ../hostap ../hostap/utils ..)
target_compile_options(pasn_oob_test PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/../pasn_endian.h)
target_compile_definitions(pasn_oob_test PRIVATE AT_NAN_BLE_OOB CONFIG_PASN CONFIG_SAE CONFIG_IEEE80211W CONFIG_ECC CONFIG_NO_RANDOM_POOL CONFIG_NO_RADIUS CONFIG_NO_VLAN)
target_link_libraries(pasn_oob_test PRIVATE green2_pasn_raw mbedcrypto)
add_test(NAME ble_pin_pasn_and_saved_pairing COMMAND pasn_oob_test)
add_test(NAME ble_pin_expired_auth3 COMMAND pasn_oob_test expire)
add_test(NAME ble_pin_replaced_auth3 COMMAND pasn_oob_test replace)
add_executable(peer_rpa_test peer_rpa_test.c ../../nimble_port/peer_rpa.c)
target_include_directories(peer_rpa_test PRIVATE ../../nimble_port)
add_test(NAME ble_resolved_peer_address COMMAND peer_rpa_test)
add_executable(peer_release_test peer_release_test.c ../nan_peer_release.c)
add_test(NAME nan_peer_mutex_ownership COMMAND peer_release_test)
add_test(NAME nan_peer_patch_boundary COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/peer_free_patch_test.py)
add_executable(ndc_reference_test ndc_reference_test.c ../nan_ndc_reference.c)
add_test(NAME nan_ndc_reference_ownership COMMAND ndc_reference_test)
