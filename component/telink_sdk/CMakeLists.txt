# Standalone TL323X vendor components. No original SDK checkout is required.
set(TELINK_COMPONENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(TELINK_PLATFORM_DIR "${TELINK_COMPONENT_ROOT}/platform")
if(TL3238_ENABLE_CS)
    set(TELINK_BLE_DIR "${TELINK_COMPONENT_ROOT}/ble/cs")
else()
    set(TELINK_BLE_DIR "${TELINK_COMPONENT_ROOT}/ble/standard")
endif()
add_library(telink_sdk_headers INTERFACE)
add_library(telink_sdk OBJECT)
target_link_libraries(telink_sdk PUBLIC telink_sdk_headers)
add_subdirectory(platform)
if(TL3238_ENABLE_BLE)
    add_subdirectory(ble)
endif()
add_subdirectory(startup)
set(TL3238_LINKER_SCRIPT "${TL3238_LINKER_SCRIPT}" PARENT_SCOPE)
