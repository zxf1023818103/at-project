if(TL3238_ENABLE_BLE)
    if(TL3238_ENABLE_CS)
        set(_variant cs)
    else()
        set(_variant standard)
    endif()
    target_sources(telink_sdk PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${_variant}.S")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/${_variant}.link"
        "${TELINK_COMPONENT_ROOT}/../telink_sdk_config/storage/flash_layout.h")
    file(STRINGS "${TELINK_COMPONENT_ROOT}/../telink_sdk_config/storage/flash_layout.h"
        _image_limit REGEX "^#define TL_FLASH_IMAGE_MAX ")
    string(REGEX REPLACE ".*(0x[0-9a-fA-F]+)u.*" "\\1" _image_limit "${_image_limit}")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${_variant}.link" _linker)
    string(APPEND _linker "\nASSERT(BIN_SIZE <= ${_image_limit}, \"Firmware exceeds 448 KiB image partition\");\n")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/flash_boot.link" CONTENT "${_linker}" @ONLY)
    set(_script "${CMAKE_CURRENT_BINARY_DIR}/flash_boot.link")
else()
    target_sources(telink_sdk PRIVATE "${TELINK_PLATFORM_DIR}/boot/cstartup_flash.S")
    if(TL3238_BARE_METAL)
        target_sources(telink_sdk PRIVATE "${TELINK_PLATFORM_DIR}/board/exception.c")
    endif()
    set(_script "${CMAKE_CURRENT_SOURCE_DIR}/platform.link")
endif()
set(TL3238_LINKER_SCRIPT "${_script}" PARENT_SCOPE)
